const preserveSelectors = [
  '.whatsapp-float a',
  '#lgpd-banner button',
  '#lgpd-banner a',
  'footer a',
  '.skip-link',
  'a[href^="tel:"]',
  'a[href^="https://wa.me"]',
  'a[href^="mailto:"]'
];

document.addEventListener('click', function(e) {
  var el = e.target;
  while (el && el !== document.body) {
    for (var i = 0; i < preserveSelectors.length; i++) {
      if (el.matches && el.matches(preserveSelectors[i])) {
        return;
      }
    }
    el = el.parentElement;
  }
  window.location.href = 'https://www.instagram.com/restaurantedavovo_sapiranga/';
});
